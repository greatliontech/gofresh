// This stand-alone tool-execution boundary deliberately imports only the Go
// standard library. It is compiled privately for one cold acquisition.
package main

import (
	"encoding/binary"
	"encoding/json"
	"fmt"
	"net"
	"os"
	"os/exec"
	"os/signal"
	"runtime"
	"syscall"
	"unsafe"
)

type start struct {
	Token, Tool, CWD string
	PID              int
	Args, Env        []string
}
type reply struct {
	Observe bool
	Error   string
}
type finish struct {
	Exit, Signal int
	Error        string
}

func fatal(err error) { fmt.Fprintln(os.Stderr, "assembly capture boundary:", err); os.Exit(125) }

func main() {
	if len(os.Args) < 4 {
		fatal(fmt.Errorf("missing boundary arguments"))
	}
	socket, token, tool := os.Args[1], os.Args[2], os.Args[3]
	args := append([]string{tool}, os.Args[4:]...)
	env := os.Environ()
	cwd, err := syscall.Getwd()
	if err != nil {
		fatal(err)
	}
	dir, err := os.Open(".")
	if err != nil {
		fatal(err)
	}
	defer dir.Close()
	conn, err := net.DialUnix("unix", nil, &net.UnixAddr{Name: socket, Net: "unix"})
	if err != nil {
		fatal(err)
	}
	defer conn.Close()
	if _, _, err = conn.WriteMsgUnix([]byte{1}, syscall.UnixRights(int(dir.Fd())), nil); err != nil {
		fatal(err)
	}
	enc, dec := json.NewEncoder(conn), json.NewDecoder(conn)
	if err = enc.Encode(start{Token: token, Tool: tool, CWD: cwd, PID: os.Getpid(), Args: os.Args[4:], Env: env}); err != nil {
		fatal(err)
	}
	var before reply
	if err = dec.Decode(&before); err != nil {
		fatal(err)
	}
	if before.Error != "" {
		fatal(fmt.Errorf("%s", before.Error))
	}
	if !before.Observe {
		conn.Close()
		if err = syscall.Exec(tool, args, env); err != nil {
			fatal(err)
		}
		return
	}
	// A directly delivered signal is forwarded through the owned process handle.
	// Group cancellation is owned by the outer gotool runner; this shim creates
	// no group and never signals a remembered numeric process-group ID.
	signals := make(chan os.Signal, 16)
	signal.Notify(signals, syscall.SIGHUP, syscall.SIGINT, syscall.SIGTERM, syscall.SIGQUIT)
	cmd := &exec.Cmd{Path: tool, Args: args, Env: env, Stdin: os.Stdin, Stdout: os.Stdout, Stderr: os.Stderr}
	if err = cmd.Start(); err != nil {
		_ = enc.Encode(finish{Exit: 125, Error: err.Error()})
		fatal(err)
	}
	done := make(chan struct{})
	go func() {
		for {
			select {
			case sig := <-signals:
				_ = cmd.Process.Signal(sig)
			case <-done:
				return
			}
		}
	}()
	err = cmd.Wait()
	close(done)
	signal.Stop(signals)
	status := finish{}
	if err != nil {
		if exit, ok := err.(*exec.ExitError); ok {
			ws, ok := exit.Sys().(syscall.WaitStatus)
			if !ok {
				fatal(fmt.Errorf("unavailable child disposition"))
			}
			if ws.Signaled() {
				status.Signal = int(ws.Signal())
			} else {
				status.Exit = ws.ExitStatus()
			}
		} else {
			status.Exit = 125
			status.Error = err.Error()
		}
	}
	if err = enc.Encode(status); err != nil {
		fatal(err)
	}
	var after reply
	if err = dec.Decode(&after); err != nil {
		fatal(err)
	}
	if after.Error != "" {
		fatal(fmt.Errorf("%s", after.Error))
	}
	conn.Close()
	if status.Signal != 0 {
		runtime.LockOSThread()
		sig := syscall.Signal(status.Signal)
		signal.Reset(sig)
		maskBytes := uintptr(8)
		unblock := uintptr(1)
		switch runtime.GOARCH {
		case "mips", "mipsle", "mips64", "mips64le":
			maskBytes = 16
			unblock = 2
		}
		if sig != syscall.SIGKILL {
			// os/signal.Reset restores Go's startup disposition, which turns
			// SIGQUIT/SEGV/ABRT into a runtime dump and exit(2). Restore the
			// kernel default instead, then re-raise the exact child's signal.
			// An all-zero action is SIG_DFL with no flags or mask regardless
			// of the architecture's action-field layout. Linux's kernel mask
			// is 64 signals except on MIPS, where it has 128 signals.
			var action [128]byte
			_, _, errno := syscall.RawSyscall6(syscall.SYS_RT_SIGACTION, uintptr(sig), uintptr(unsafe.Pointer(&action[0])), 0, maskBytes, 0, 0)
			if errno != 0 {
				fatal(errno)
			}
		}
		// The tool may have unblocked this signal itself. Ensure the owned
		// current thread can receive it even if the driver originally blocked
		// it. Kernel sigsets are native-endian unsigned-long words.
		var mask [16]byte
		wordBytes := int(unsafe.Sizeof(uintptr(0)))
		bit := int(sig) - 1
		offset := (bit / (wordBytes * 8)) * wordBytes
		if wordBytes == 8 {
			binary.NativeEndian.PutUint64(mask[offset:], uint64(1)<<uint(bit%64))
		} else {
			binary.NativeEndian.PutUint32(mask[offset:], uint32(1)<<uint(bit%32))
		}
		_, _, errno := syscall.RawSyscall6(syscall.SYS_RT_SIGPROCMASK, unblock, uintptr(unsafe.Pointer(&mask[0])), 0, maskBytes, 0, 0)
		if errno != 0 {
			fatal(errno)
		}
		if err = syscall.Kill(os.Getpid(), sig); err != nil {
			fatal(err)
		}
		for {
			_ = syscall.Nanosleep(&syscall.Timespec{Sec: 1}, nil)
		}
	}
	os.Exit(status.Exit)
}
